// Added protocol whitelist validation
const SAFE_PROTOCOLS = ['http:', 'https:', 'about:'];
try {
  const parsedUrl = new URL(url);
  if (!SAFE_PROTOCOLS.includes(parsedUrl.protocol)) {
    console.error(`❌ Blocked unsafe protocol: ${parsedUrl.protocol}`);

**Existing Code:**
```typescript
function isAllowedPluginSource(source: string): boolean {
  if (!source || typeof source !== 'string') return false;
  // Allow HTTPS URLs from trusted origins
  if (source.startsWith('https://')) {
    return ALLOWED_PLUGIN_ORIGINS.some(origin => source.startsWith(origin));
  }
  // Allow local file paths only within userData plugins dir
  const pluginsDir = path.join(app.getPath('userData'), 'plugins');
  const resolved = path.resolve(source);
  return resolved.startsWith(pluginsDir + path.sep) || resolved === pluginsDir;
}
```

**Status:** Whitelist already active, restricts plugin sources to:
- HTTPS: `marketplace.zeno-browser.com`, `plugins.mybonzo.com`
- Local: `app.getPath('userData')/plugins/` only

### CR-003: Plugin Code Execution Security
**File:** `src/plugin-system/core/plugin-loader.ts`  
**Lines:** 1-6 (added `import * as vm from 'vm'`), 73-107 (replaced `executeSandboxed()` method)

**Change:**
```typescript
// BEFORE: Dangerous new Function()
private executeSandboxed(code: string): any {
  const fn = new Function('BasePlugin', `
    "use strict";
    ${code}
    return { default: exports.default || null };
  `);
  return fn(BasePlugin);
}

// AFTER: Secure vm.runInNewContext()
import * as vm from 'vm';

private executeSandboxed(code: string): any {
  const sandbox: {
    BasePlugin: typeof BasePlugin;
    exports: { default?: any };
    console: any;
  } = {
    BasePlugin,
    exports: {},
    console: {
      log: (...args: any[]) => console.log('[Plugin]', ...args),
      // ...
    },
  };

  const wrappedCode = `
    "use strict";
    ${code}
    exports.default;
  `;

  const result = vm.runInNewContext(wrappedCode, sandbox, {
    timeout: 5000, // 5s timeout
    displayErrors: true,
  });

  return { default: result || sandbox.exports.default };
}
```

**Impact:**
- Plugin code runs in isolated VM context (not global scope)
- 5-second execution timeout prevents infinite loops
- No access to `process`, `require`, `__dirname`, etc.
- Limited console methods only (log, warn, error with `[Plugin]` prefix)
