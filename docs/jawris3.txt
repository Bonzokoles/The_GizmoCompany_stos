## 🎨 `models/timezone_model.dart`

```dart
import 'package:hive/hive.dart';

part 'timezone_model.g.dart';

@HiveType(typeId: 0)
class Timezone extends HiveObject {
  @HiveField(0)
  final String id;

  @HiveField(1)
  final String name;

  @HiveField(2)
  final String tzName;

  @HiveField(3)
  final bool isPinned;

  @HiveField(4)
  final DateTime addedAt;

  Timezone({
    required this.id,
    required this.name,
    required this.tzName,
    this.isPinned = false,
    required this.addedAt,
  });

  Timezone copyWith({
    String? id,
    String? name,
    String? tzName,
    bool? isPinned,
    DateTime? addedAt,
  }) {
    return Timezone(
      id: id ?? this.id,
      name: name ?? this.name,
      tzName: tzName ?? this.tzName,
      isPinned: isPinned ?? this.isPinned,
      addedAt: addedAt ?? this.addedAt,
    );
  }

  @override
  String toString() => 'Timezone(name: $name, tz: $tzName)';
}

class TimezoneAdapter extends TypeAdapter<Timezone> {
  @override
  final typeId = 0;

  @override
  Timezone read(BinaryReader reader) {
    return Timezone(
      id: reader.readString(),
      name: reader.readString(),
      tzName: reader.readString(),
      isPinned: reader.readBool(),
      addedAt: DateTime.parse(reader.readString()),
    );
  }

  @override
  void write(BinaryWriter writer, Timezone obj) {
    writer.writeString(obj.id);
    writer.writeString(obj.name);
    writer.writeString(obj.tzName);
    writer.writeBool(obj.isPinned);
    writer.writeString(obj.addedAt.toIso8601String());
  }
}
```

## 🔄 `providers/app_provider.dart` - STATE MANAGEMENT

```dart
import 'package:flutter/material.dart';
import 'package:hive/hive.dart';
import 'package:uuid/uuid.dart';
import 'package:timezone/timezone.dart' as tz;
import '../models/timezone_model.dart';

class AppProvider extends ChangeNotifier {
  late Box<Timezone> _timezoneBox;
  late Box _settingsBox;
  late List<Timezone> _timezones;
  bool _format24h = true;
  bool _isDarkTheme = true;
  bool _showDate = true;
  bool _showUTCOffset = true;
  bool _autoRefresh = true;
  late DateTime _currentTime;

  late Timer _timer;

  AppProvider() {
    _initProvider();
  }

  Future<void> _initProvider() async {
    _timezoneBox = Hive.box<Timezone>('timezones');
    _settingsBox = Hive.box('settings');
    _currentTime = DateTime.now();

    // Ładuj ustawienia
    _format24h = _settingsBox.get('format24h', defaultValue: true);
    _isDarkTheme = _settingsBox.get('isDarkTheme', defaultValue: true);
    _showDate = _settingsBox.get('showDate', defaultValue: true);
    _showUTCOffset = _settingsBox.get('showUTCOffset', defaultValue: true);

    // Ładuj strefy czasowe
    _loadTimezones();

    // Jeśli puste, dodaj domyślne
    if (_timezones.isEmpty) {
      _addDefaultTimezones();
    }

    // Uruchom auto-refresh
    _startAutoRefresh();
    notifyListeners();
  }

  void _startAutoRefresh() {
    _timer = Timer.periodic(const Duration(seconds: 1), (_) {
      _currentTime = DateTime.now();
      notifyListeners();
    });
  }

  void _loadTimezones() {
    _timezones = _timezoneBox.values.toList();
  }

  void _addDefaultTimezones() {
    const defaultTzs = [
      ('America/New_York', 'New York'),
      ('America/Los_Angeles', 'Los Angeles'),
      ('Europe/London', 'London'),
      ('Europe/Paris', 'Paris'),
      ('Asia/Tokyo', 'Tokyo'),
      ('Asia/Shanghai', 'Shanghai'),
      ('Australia/Sydney', 'Sydney'),
      ('Asia/Dubai', 'Dubai'),
      ('America/Sao_Paulo', 'São Paulo'),
      ('Asia/Kolkata', 'India'),
    ];

    for (var (tzName, displayName) in defaultTzs) {
      addTimezone(displayName, tzName);
    }
  }

  // ========== Getters ==========
  DateTime get currentTime => _currentTime;
  List<Timezone> get timezones => _timezones;
  bool get format24h => _format24h;
  bool get isDarkTheme => _isDarkTheme;
  bool get showDate => _showDate;
  bool get showUTCOffset => _showUTCOffset;

  // ========== Timezone Methods ==========
  Future<void> addTimezone(String displayName, String tzName) async {
    const uuid = Uuid();
    final timezone = Timezone(
      id: uuid.v4(),
      name: displayName,
      tzName: tzName,
      addedAt: DateTime.now(),
    );

    await _timezoneBox.add(timezone);
    _timezones.add(timezone);
    notifyListeners();
  }

  Future<void> removeTimezone(String id) async {
    final index = _timezones.indexWhere((t) => t.id == id);
    if (index != -1) {
      await _timezoneBox.deleteAt(index);
      _timezones.removeAt(index);
      notifyListeners();
    }
  }

  Future<void> togglePinTimezone(String id) async {
    final index = _timezones.indexWhere((t) => t.id == id);
    if (index != -1) {
      final timezone = _timezones[index];
      _timezones[index] = timezone.copyWith(isPinned: !timezone.isPinned);
      await _timezoneBox.putAt(index, _timezones[index]);
      notifyListeners();
    }
  }

  // ========== Settings Methods ==========
  Future<void> setFormat24h(bool value) async {
    _format24h = value;
    await _settingsBox.put('format24h', value);
    notifyListeners();
  }

  Future<void> setDarkTheme(bool value) async {
    _isDarkTheme = value;
    await _settingsBox.put('isDarkTheme', value);
    notifyListeners();
  }

  Future<void> setShowDate(bool value) async {
    _showDate = value;
    await _settingsBox.put('showDate', value);
    notifyListeners();
  }

  Future<void> setShowUTCOffset(bool value) async {
    _showUTCOffset = value;
    await _settingsBox.put('showUTCOffset', value);
    notifyListeners();
  }

  // ========== Utility Methods ==========
  String getTimeForTimezone(Timezone timezone) {
    try {
      final location = tz.getLocation(timezone.tzName);
      final now = tz.TZDateTime.now(location);

      if (_format24h) {
        return '${now.hour.toString().padLeft(2, '0')}:${now.minute.toString().padLeft(2, '0')}:${now.second.toString().padLeft(2, '0')}';
      } else {
        final hour = now.hour % 12 == 0 ? 12 : now.hour % 12;
        final ampm = now.hour >= 12 ? 'PM' : 'AM';
        return '${hour.toString().padLeft(2, '0')}:${now.minute.toString().padLeft(2, '0')}:${now.second.toString().padLeft(2, '0')} $ampm';
      }
    } catch (e) {
      return 'ERROR';
    }
  }

  String getDateForTimezone(Timezone timezone) {
    try {
      final location = tz.getLocation(timezone.tzName);
      final now = tz.TZDateTime.now(location);
      return '${_dayName(now.weekday)}, ${_monthName(now.month)} ${now.day}, ${now.year}';
    } catch (e) {
      return '';
    }
  }

  String getUTCOffsetForTimezone(Timezone timezone) {
    try {
      final location = tz.getLocation(timezone.tzName);
      final now = tz.TZDateTime.now(location);
      final offset = now.timeZoneOffset;
      final hours = offset.inHours;
      final minutes = (offset.inMinutes % 60).abs();
      final sign = hours >= 0 ? '+' : '-';
      return 'UTC$sign${hours.abs().toString().padLeft(2, '0')}:${minutes.toString().padLeft(2, '0')}';
    } catch (e) {
      return 'UTC';
    }
  }

  String _dayName(int day) {
    const days = ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'];
    return days[day - 1];
  }

  String _monthName(int month) {
    const months = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
    return months[month - 1];
  }

  @override
  void dispose() {
    _timer.cancel();
    super.dispose();
  }
}
```

## 🔗 `services/timezone_service.dart`

```dart
import 'package:timezone/timezone.dart' as tz;

class TimezoneService {
  static List<String> getAllTimezones() {
    return tz.timeZoneDatabase.locations.keys.toList()..sort();
  }

  static List<String> getPopularTimezones() {
    return [
      'America/New_York',
      'America/Los_Angeles',
      'America/Chicago',
      'America/Denver',
      'Europe/London',
      'Europe/Paris',
      'Europe/Berlin',
      'Europe/Rome',
      'Asia/Tokyo',
      'Asia/Shanghai',
      'Asia/Hong_Kong',
      'Asia/Singapore',
      'Asia/Dubai',
      'Asia/Bangkok',
      'Asia/Kolkata',
      'Australia/Sydney',
      'Australia/Melbourne',
      'Pacific/Auckland',
      'America/Sao_Paulo',
      'Africa/Cairo',
      'Africa/Johannesburg',
    ];
  }

  static String formatTime(DateTime time, bool format24h) {
    if (format24h) {
      return '${time.hour.toString().padLeft(2, '0')}:${time.minute.toString().padLeft(2, '0')}:${time.second.toString().padLeft(2, '0')}';
    } else {
      final hour = time.hour % 12 == 0 ? 12 : time.hour % 12;
      final ampm = time.hour >= 12 ? 'PM' : 'AM';
      return '${hour.toString().padLeft(2, '0')}:${time.minute.toString().padLeft(2, '0')}:${time.second.toString().padLeft(2, '0')} $ampm';
    }
  }

  static String formatDate(DateTime date) {
    const days = ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'];
    const months = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
    
    return '${days[date.weekday - 1]}, ${months[date.month - 1]} ${date.day}, ${date.year}';
  }

  static String getUTCOffset(DateTime date) {
    final offset = date.timeZoneOffset;
    final hours = offset.inHours;
    final minutes = (offset.inMinutes % 60).abs();
    final sign = hours >= 0 ? '+' : '-';
    return 'UTC$sign${hours.abs().toString().padLeft(2, '0')}:${minutes.toString().padLeft(2, '0')}';
  }
}
```

```dart
import 'package:flutter_test/flutter_test.dart';
import 'package:timezone/timezone.dart' as tz;

void main() {
  group('TimezoneService Tests', () {

      // Initialize timezone data
    });

    test('Get all timezones', () {
      final timezones = TimezoneService.getAllTimezones();
      expect(timezones.isNotEmpty, true);
      expect(timezones.contains('America/New_York'), true);
    });

    test('Get popular timezones', () {
      final popular = TimezoneService.getPopularTimezones();
      expect(popular.length, 21);
      expect(popular.contains('America/New_York'), true);
      expect(popular.contains('Asia/Tokyo'), true);
    });

    test('Format time 24h', () {
      final time = DateTime(2024, 1, 15, 14, 30, 45);
      final formatted = TimezoneService.formatTime(time, true);
      expect(formatted, '14:30:45');
    });

    test('Format time 12h', () {
      final time = DateTime(2024, 1, 15, 14, 30, 45);
      final formatted = TimezoneService.formatTime(time, false);
      expect(formatted.contains('PM'), true);
    });

      final date = DateTime(2024, 1, 15);
      final formatted = TimezoneService.formatDate(date);
      expect(formatted.contains('Jan'), true);
      expect(formatted.contains('15'), true);
    });
```
